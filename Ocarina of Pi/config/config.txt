# Place this file at the root of the SD card boot partition

gpu_freq=250

sdram_freq=450

# NTSC artifact 240p mode (use 16 for PAL 240p)

# === Memory Split ===
# 256MB GPU (required for VideoCore IV + texture streaming)

# === Power Management ===
max_usb_current=1
power_off_on_halt=0

# === Performance ===
# Disable activity LED blinking
dtparam=act_led_trigger=none
dtparam=act_led_activelow=off

# === Audio ===
# Force 3.5mm TRRS audio output (not HDMI)
audio_pwm_mode=2
audio_pwm_boost=5
hdmi_ignore_cec_init=1

# === Kernel / Boot ===
disable_splash=1
avoid_warnings=1
boot_delay=1

# === Overclock warnings ===
arm_freq_min=200
kernel=kernel7.img

# === Optional: Reduce GPU memory fragmentation ===
dispmanx_offline=1

# === Disable unnecessary hardware ===
dtparam=i2c_arm=off

dtparam=audio=off
